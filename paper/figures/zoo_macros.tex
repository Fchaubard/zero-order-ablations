% 1.5-INP (added after the protocol): INP's colour, dashed, plotted last with explicit options.
\pgfplotsset{zooinpb/.style={color=zooinp, densely dashed, mark=pentagon, mark size=1.5pt}}

% Seed band (min..max over seeds) of one method: #1 csv, #2 column prefix, #3 colour.
\newcommand{\zooband}[3]{%
  \addplot [name path=#2max, draw=none, forget plot] table [x=time, y=#2_max, col sep=comma] {#1};
  \addplot [name path=#2min, draw=none, forget plot] table [x=time, y=#2_min, col sep=comma] {#1};
  \addplot [#3!30, forget plot] fill between [of=#2max and #2min];
}
