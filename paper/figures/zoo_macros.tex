% Shared pgfplots macros for the stage-2 figures (input from the preamble of main.tex).
\usepgfplotslibrary{fillbetween}

% Probe-method cycle list (SPSA, RS, NP, ANP, INP, DANP) = zoocycle without BP and FF.
\pgfplotscreateplotcyclelist{zooprobecycle}{%
  color=zoospsa, solid, mark=o,         mark size=1.3pt\\
  color=zoors,   solid, mark=square*,   mark size=1.2pt\\
  color=zoonp,   solid, mark=triangle*, mark size=1.5pt\\
  color=zooanp,  solid, mark=diamond*,  mark size=1.5pt\\
  color=zooinp,  solid, mark=pentagon*, mark size=1.5pt\\
  color=zoodanp, solid, mark=x,         mark size=1.8pt\\
}

% Seed band (min..max over seeds) of one method: #1 csv, #2 column prefix, #3 colour.
% The two boundary paths are invisible and kept out of the legend.
\newcommand{\zooband}[3]{%
  \addplot [name path=#2max, draw=none, forget plot] table [x=time, y=#2_max, col sep=comma] {#1};
  \addplot [name path=#2min, draw=none, forget plot] table [x=time, y=#2_min, col sep=comma] {#1};
  \addplot [#3!30, forget plot] fill between [of=#2max and #2min];
}
% All eight bands of a curve2_{task}.csv file.
\newcommand{\zoobands}[1]{%
  \zooband{#1}{bp}{black}
  \zooband{#1}{spsa}{zoospsa}
  \zooband{#1}{rs}{zoors}
  \zooband{#1}{np}{zoonp}
  \zooband{#1}{anp}{zooanp}
  \zooband{#1}{inp}{zooinp}
  \zooband{#1}{danp}{zoodanp}
  \zooband{#1}{ff}{zooff}
}
% The eight seed-mean curves of a curve2_{task}.csv file (zoocycle order, marks every 10 min).
\newcommand{\zoomeans}[1]{%
  \addplot table [x=time, y=bp_mean,   col sep=comma] {#1};
  \addplot table [x=time, y=spsa_mean, col sep=comma] {#1};
  \addplot table [x=time, y=rs_mean,   col sep=comma] {#1};
  \addplot table [x=time, y=np_mean,   col sep=comma] {#1};
  \addplot table [x=time, y=anp_mean,  col sep=comma] {#1};
  \addplot table [x=time, y=inp_mean,  col sep=comma] {#1};
  \addplot table [x=time, y=danp_mean, col sep=comma] {#1};
  \addplot table [x=time, y=ff_mean,   col sep=comma] {#1};
}
